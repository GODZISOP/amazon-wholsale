-- Run once in Supabase SQL Editor:
create table if not exists portfolio_config (
  id text primary key,
  data jsonb not null,
  updated_at timestamptz default now()
);
alter table portfolio_config enable row level security;
grant all on table public.portfolio_config to anon, authenticated;
drop policy if exists "Public read" on portfolio_config;
drop policy if exists "Public upsert" on portfolio_config;
create policy "Public read" on portfolio_config for select using (true);
create policy "Public upsert" on portfolio_config for all using (true) with check (true);

-- Storage bucket + upload permissions:
insert into storage.buckets (id, name, public)
values ('portfolio-assets', 'portfolio-assets', true)
on conflict (id) do update set public = true;

drop policy if exists "Public bucket read" on storage.objects;
drop policy if exists "Public bucket upload" on storage.objects;
drop policy if exists "Public bucket update" on storage.objects;
create policy "Public bucket read" on storage.objects for select using (bucket_id = 'portfolio-assets');
create policy "Public bucket upload" on storage.objects for insert with check (bucket_id = 'portfolio-assets');
create policy "Public bucket update" on storage.objects for update using (bucket_id = 'portfolio-assets');
